# CrocGUI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A lightweight Go + Bubble Tea terminal GUI for the croc file-transfer CLI: a welcome screen, a Send flow (native Windows multi-file dialog → share code + live progress), a Receive flow (code input → live progress into `Downloads`), and automatic install of croc via winget when it's missing.

**Architecture:** A thin TUI wraps the `croc` CLI as a subprocess. The `croc` package is pure and UI-free: binary detection, argument construction, and output parsing. The `ui` package owns the Bubble Tea screens and runs the subprocess via `tea.Exec`, dispatching each parsed output line to the model as a tea message. The `dialog` package bridges to a native Windows file dialog through PowerShell. Nothing is embedded from croc's library — we always shell out.

**Tech Stack:** Go 1.26+, `github.com/charmbracelet/bubbletea` (v2), `github.com/charmbracelet/bubbles` (textinput, progress, spinner), `github.com/charmbracelet/lipgloss`. No other runtime dependencies.

## Global Constraints

- Module name is `crocgui`; run `go mod init crocgui` in the repo root.
- Go 1.26+ (installed: `go1.26.5`).
- External dependencies are limited to: `github.com/charmbracelet/bubbletea`, `github.com/charmbracelet/bubbles`, `github.com/charmbracelet/lipgloss`.
- croc is always driven as a subprocess — never imported as a Go library.
- **croc v10 CLI facts (verified against 10.6.0):**
  - Global flags precede the subcommand: `croc --yes send <files…>`.
  - `--pass` is the RELAY password (default `pass123`), NOT the transfer code.
  - The transfer code is the bare argument on receive; on send it is generated by croc and printed as `Code is: <code>`.
  - `send` subcommand flags: `--code`/`-c` (min 6 chars), `--zip`, `--text`, `--git`, `--no-local`.
  - Global receive options: `--out <dir>` (output folder, default `.`), `--rename` (receive under a new name without prompting), `--yes` (auto-agree to all prompts).
- Exact subprocess invocations:
  - Send: `croc --yes send <file1> <file2> …`
  - Receive: `croc --yes --rename <code> --out <downloadsDir>`
- croc auto-copies the generated code to the clipboard and prints `Code copied to clipboard!` — the app relies on this; it adds no clipboard code.
- The native file dialog is Windows-only (PowerShell + `System.Windows.Forms`); the rest of the app is portable.
- TDD: write a failing test first, then the minimal implementation, then commit. Small frequent commits.
- Spec: `docs/superpowers/specs/2026-08-05-crocgui-design.md` (screen mockups live there; reference it, don't duplicate).

```go
package ui

import tea "github.com/charmbracelet/bubbletea"

type screen int

const (
	screenWelcome screen = iota
	screenSend
	screenReceive
	screenInstall
)

// goScreenMsg asks the root model to switch to screen s.
type goScreenMsg struct{ s screen }

// rootModel dispatches to the active screen model.
type rootModel struct {
	cur   screen
	bin   string
	wel   welcomeModel
	send  sendModel
	recv  receiveModel
	inst  installModel
}

// New returns the root model. If installErr != nil the app starts on the
// install screen (croc is missing).
func New(bin string, installErr error) tea.Model {
	r := &rootModel{cur: screenWelcome, bin: bin, wel: welcomeModel{selected: 0}}
	if installErr != nil {
		r.cur = screenInstall
		r.inst = newInstallModel()
	}
	return r
}

func (r *rootModel) Init() tea.Cmd {
	switch r.cur {
	case screenWelcome:
		return r.wel.Init()
	case screenSend:
		return r.send.Init()
	case screenReceive:
		return r.recv.Init()
	case screenInstall:
		return r.inst.Init()
	}
	return nil
}

func (r *rootModel) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	switch msg := msg.(type) {
	case goScreenMsg:
		r.cur = msg.s
		r.bin = "" // cleared; re-set on route entry (see Tasks 4-6)
		return r, r.Init()
	}
	switch r.cur {
	case screenWelcome:
		var cmd tea.Cmd
		r.wel, cmd = updateWelcome(r.wel, msg)
		return r, cmd
	}
	return r, nil
}

func (r *rootModel) View() string {
	switch r.cur {
	case screenWelcome:
		return r.wel.View()
	}
	return ""
}
```

`updateWelcome` is a helper so the root can keep a copy of the model (`welcomeModel.Update` returns `tea.Model`; we type-assert back). Define it in `ui/welcome.go`.

```go
package ui

import (
	"github.com/charmbracelet/lipgloss"
	tea "github.com/charmbracelet/bubbletea"
)

var (
	welcomeStyle  = lipgloss.NewStyle().Bold(true).Foreground(lipgloss.Color("32"))
	dimStyle      = lipgloss.NewStyle().Foreground(lipgloss.Color("8"))
	selectedStyle = lipgloss.NewStyle().Reverse(true)
)

type welcomeModel struct {
	selected int // 0 = Send, 1 = Receive
}

func (w welcomeModel) Init() tea.Cmd { return nil }

func updateWelcome(w welcomeModel, msg tea.Msg) (welcomeModel, tea.Cmd) {
	switch msg := msg.(type) {
	case tea.KeyMsg:
		switch msg.String() {
		case "up", "left", "k", "h", "tab":
			w.selected = 1 - w.selected
		case "down", "right", "j", "l", "shift+tab":
			w.selected = 1 - w.selected
		case "enter", "s", "r":
			if msg.String() == "r" {
				w.selected = 1
			} else if msg.String() == "s" {
				w.selected = 0
			}
			if w.selected == 0 {
				return w, func() tea.Msg { return goScreenMsg{s: screenSend} }
			}
			return w, func() tea.Msg { return goScreenMsg{s: screenReceive} }
		case "q", "ctrl+c":
			return w, tea.Quit
		}
	}
	return w, nil
}

func (w welcomeModel) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	m, cmd := updateWelcome(w, msg)
	return m, cmd
}

func (w welcomeModel) View() string {
	send := "  Send  "
	recv := "  Receive  "
	if w.selected == 0 {
		send = selectedStyle.Render(send)
	} else {
		recv = selectedStyle.Render(recv)
	}
	line := lipgloss.JoinHorizontal(lipgloss.Center, "  ", send, recv, "  ")
	return lipgloss.JoinVertical(lipgloss.Center,
		welcomeStyle.Render("CROC"),
		dimStyle.Render("easily & securely transfer files\npeer-to-peer over the internet"),
		"",
		line,
		"",
		dimStyle.Render("s / arrows select · enter confirm · q / ctrl-c quit"),
	) + "\n"
}
```

```go
package croc

import (
	"regexp"
	"strconv"
	"strings"
)

type EventKind int

const (
	EventStatus EventKind = iota
	EventCode
	EventFile
	EventProgress
	EventError
)

type Event struct {
	Kind    EventKind
	Code    string
	File    string
	Rate    string
	Text    string
	Percent float64 // -1 when unknown
}

var (
	codeRe    = regexp.MustCompile(`Code is:\s+(\S+)`)
	fileRe    = regexp.MustCompile(`Sending\s+'([^']+)'`)
	errorRe   = regexp.MustCompile(`(?i)(could not connect|bad response|error:|failed)`)
	percentRe = regexp.MustCompile(`(\d+(?:\.\d+)?)\s*%`)
	rateRe    = regexp.MustCompile(`(\d+(?:\.\d+)?)\s*[KMGT]?B/s`)
)

// ParseLine parses a single line/fragment of croc output into an Event.
func ParseLine(line string) Event {
	if code := codeRe.FindStringSubmatch(line); code != nil {
		return Event{Kind: EventCode, Code: code[1], Text: line}
	}
	if f := fileRe.FindStringSubmatch(line); f != nil {
		return Event{Kind: EventFile, File: f[1], Text: line}
	}
	if errorRe.MatchString(line) {
		return Event{Kind: EventError, Text: line}
	}
	if p := percentRe.FindStringSubmatch(line); p != nil {
		pct, _ := strconv.ParseFloat(p[1], 64)
		rate := ""
		if r := rateRe.FindString(line); r != "" {
			rate = r
		}
		return Event{Kind: EventProgress, Percent: pct, Rate: rate, Text: line}
	}
	return Event{Kind: EventStatus, Text: line}
}

// ParseChunk splits raw output on \r and \n, drops blank/whitespace
// fragments, and returns an Event per fragment.
func ParseChunk(chunk string) []Event {
	var out []Event
	for _, frag := range strings.FieldsFunc(chunk, func(r rune) bool { return r == '\r' || r == '\n' }) {
		if strings.TrimSpace(frag) == "" {
			continue
		}
		out = append(out, ParseLine(frag))
	}
	return out
}
```

```go
package dialog

import (
	"bufio"
	"os/exec"
	"strings"
)

// dialogScript is a PowerShell fragment using WinForms to show a native,
// multi-select file dialog. Output is UTF-8, one absolute path per line.
// Empty output means the user cancelled.
func dialogScript() string {
	return `[Console]::OutputEncoding = [System.Text.Encoding]::UTF8
Add-Type -AssemblyName System.Windows.Forms
$dialog = New-Object System.Windows.Forms.OpenFileDialog
$dialog.Multiselect = $true
$dialog.Title = 'Select files to send'
if ($dialog.ShowDialog() -eq [System.Windows.Forms.DialogResult]::OK) {
    $dialog.FileNames | ForEach-Object { $_.ToString() }
}`
}

// PickFiles opens a native Windows multi-select dialog and returns the chosen
// absolute paths. An empty slice means the user cancelled.
func PickFiles() ([]string, error) {
	// -STA is required for WinForms dialogs on Windows.
	cmd := exec.Command("powershell.exe", "-NoProfile", "-STA", "-Command", dialogScript())
	out, err := cmd.Output()
	if err != nil {
		return nil, err
	}
	var files []string
```

```go
// formatSize renders a byte count in a compact human form.
func formatSize(n int64) string {
	const unit = 1024
	if n < unit {
		return fmt.Sprintf("%d B", n)
	}
	div, exp := int64(unit), 0
	for m := n / unit; m >= unit; m /= unit {
		div *= unit
		exp++
	}
	return fmt.Sprintf("%.1f %cB", float64(n)/float64(div), "KMGTPE"[exp])
}

// downloadsDir returns the user's Downloads folder, creating it if needed.
func downloadsDir() (string, error) {
	home, err := os.UserHomeDir()
	if err != nil {
		return "", err
	}
	d := filepath.Join(home, "Downloads")
	if err := os.MkdirAll(d, 0o755); err != nil {
		return "", err
	}
	return d, nil
}
```

```go
// crocEventMsg carries one parsed line of croc output to the model.
type crocEventMsg struct{ ev croc.Event }

// crocFinishedMsg is delivered when the croc subprocess exits.
type crocFinishedMsg struct{ err error }

// buildCrocCmd constructs the subprocess command with stdout+stderr wired to a
// shared pipe so output can be streamed back as messages. Kept pure for tests.
func buildCrocCmd(bin string, args []string, dir string) *exec.Cmd {
	cmd := exec.Command(bin, args...)
	cmd.Dir = dir
	r, w := io.Pipe()
	cmd.Stdout = w
	cmd.Stderr = w
	// TODO: read from r in a goroutine inside runCroc
	return cmd
}

// runCroc starts croc and returns a tea.Cmd that completes when croc exits.
// While running, each output line is parsed and dispatched as crocEventMsg.
func runCroc(p *tea.Program, bin string, args []string, dir string) tea.Cmd {
	cmd := exec.Command(bin, args...)
	cmd.Dir = dir
	r, w := io.Pipe()
	cmd.Stdout = w
	cmd.Stderr = w

	// Stream output: split on \r and \n so progress frames arrive live.
	go func() {
		defer r.Close()
		br := bufio.NewReader(r)
		var frag []byte
		flush := func() {
			if len(strings.TrimSpace(string(frag))) == 0 {
				frag = frag[:0]
				return
			}
			p.Send(crocEventMsg{ev: croc.ParseLine(string(frag))})
			frag = frag[:0]
		}
		for {
			b, err := br.ReadByte()
			if err != nil {
				break
			}
			if b == '\r' || b == '\n' {
				flush()
				continue
			}
			frag = append(frag, b)
		}
		flush()
	}()

	return tea.Exec(cmd, func(err error) tea.Msg {
		w.Close()
		return crocFinishedMsg{err: err}
	})
}
```

```go
	switch m.state {
	case recvInput:
		if k, ok := msg.(tea.KeyMsg); ok {
			if k.String() == "enter" {
				code := m.code.Value()
				if len(code) < 6 {
					m.status = "code must be at least 6 characters"
					return m, nil
				}
				m.state = recvWaiting
				return m, tea.Batch(m.spinner.Tick, func() tea.Msg {
					return goRunReceiveMsg{m}
				})
			}
			if k.String() == "esc" {
				return m, func() tea.Msg { return goScreenMsg{s: screenWelcome} }
			}
		}
		var cmd tea.Cmd
		m.code, cmd = m.code.Update(msg)
		return m, cmd

	case recvWaiting, recvTransferring:
		var cmd tea.Cmd
		m.spinner, cmd = m.spinner.Update(msg)
		if cmd != nil {
			return m, cmd
		}
		switch msg := msg.(type) {
		case crocEventMsg:
			ev := msg.ev
			switch ev.Kind {
			case croc.EventFile:
				m.file = ev.File
			case croc.EventProgress:
				m.pct = ev.Percent
				m.rate = ev.Rate
				m.state = recvTransferring
			case croc.EventError:
				m.status = ev.Text
				m.state = recvError
			default:
				if ev.Text != "" {
					m.status = ev.Text
				}
			}
			return m, nil
		case crocFinishedMsg:
			if msg.err != nil {
				m.err = msg.err
				m.state = recvError
			} else {
				m.state = recvDone
			}
			return m, nil
		case tea.KeyMsg:
			if k := msg.String(); k == "q" || k == "ctrl+c" {
				return m, tea.Quit
			}
		}
		return m, nil
	}
	return m, nil
}
```

Wait — a receiveModel can't receive `p *tea.Program` from nowhere; the root sets it. To keep models constructible and testable, `runCroc` is invoked from the root, not from inside the receive model. The cleanest flow:

- The receive screen, on Enter, returns a `tea.Cmd` that captures the program pointer via the `tea.WindowSizeMsg`/program plumbing. Bubble Tea v2 does NOT hand the program to models. The standard way to get `*tea.Program` inside a model is to set it after construction:

```go
p := tea.NewProgram(m)
// after run:
if mm, ok := m.(*rootModel); ok { mm.setProgram(p) }
```

To keep this plan simple and idiomatic, the root owns the `*tea.Program` and builds the receive/send models with it. In `main.go`:

```go
p := tea.NewProgram(m, tea.WithAltScreen())
if rm, ok := m.(*rootModel); ok {
	rm.prog = p
}
if _, err := p.Run(); err != nil { ... }
```

So `rootModel` gains a `prog *tea.Program` field, and `New(bin, err)` seeds it (nil until `main` sets it). `runCroc` is called from the root's `Update` when the receive model reports it wants to start.

This keeps each screen model free of the program pointer and testable in isolation. The receive screen's Enter handler emits:

```go
package ui

import (
	"github.com/charmbracelet/bubbles/spinner"
	tea "github.com/charmbracelet/bubbletea"

	"crocgui/croc"
)

type installState int

const (
	installPrompt installState = iota
	installRunning
	installDone
	installFailed
)

type installModel struct {
	state   installState
	spinner spinner.Model
	err     error
}

func newInstallModel() installModel {
	return installModel{
		state:   installPrompt,
		spinner: spinner.New(spinner.WithSpinner(spinner.Dot)),
	}
}

type installStartedMsg struct{}
type installFinishedMsg struct{ err error }

func (m installModel) Init() tea.Cmd { return nil }

func (m installModel) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	switch m.state {
	case installPrompt:
		if k, ok := msg.(tea.KeyMsg); ok {
			switch k.String() {
			case "enter", "y":
				return m, func() tea.Msg { return installStartedMsg{} }
			case "q", "ctrl+c", "n":
				return m, tea.Quit
			}
		}
		return m, nil
	default:
		switch msg := msg.(type) {
		case installStartedMsg:
			m.state = installRunning
			return m, tea.Batch(m.spinner.Tick, func() tea.Msg {
				return installFinishedMsg{err: croc.Install()}
			})
		case installFinishedMsg:
			if msg.err != nil {
				m.state = installFailed
				m.err = msg.err
			} else {
				m.state = installDone
			}
			return m, nil
		case tea.KeyMsg:
			if m.state == installDone {
				// re-detect and move to welcome
				if bin, err := croc.Detect(); err == nil {
					return m, func() tea.Msg { return goScreenMsg{s: screenWelcome, bin: bin} }
				}
			}
			if k := msg.String(); k == "q" || k == "ctrl+c" {
				return m, tea.Quit
			}
		}
		var cmd tea.Cmd
		m.spinner, cmd = m.spinner.Update(msg)
		return m, cmd
	}
}

func (m installModel) View() string {
	switch m.state {
	case installPrompt:
		return lipgloss.NewStyle().Padding(1).Render(
			"croc is not installed.\n\n" +
				"Install it via winget?\n\n" +
				"  [Y]es   [N]o (quit)\n\n" +
				dimStyle.Render("This will run: winget install schollz.croc"),
		) + "\n"
	case installRunning:
		return m.spinner.View() + " Installing croc via winget…\n"
	case installDone:
		return "croc installed successfully! Press Enter to continue.\n"
	case installFailed:
		return "Failed to install croc:\n" + m.err.Error() + "\n\nPress Esc to quit.\n"
	}
	return ""
}
```
